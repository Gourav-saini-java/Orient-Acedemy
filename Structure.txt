│   ├── adminlogout.php       # Admin logout handler
│   ├── admin_auth.php        # Admin route protection middleware
│   ├── admindashboard.php    # Central admin KPI metrics & quick hub
│   ├── managecourses.php     # Course catalog management
│   ├── addcourse.php         # Add new course
│   ├── editcourse.php        # Edit course details

│   └── reports.php           # Visual analytics & Chart.js reports

│   ├── header.php            # HTML head, typography, Bootstrap CSS
│   ├── navbar.php            # Responsive navbar with Orient Academy branding
│   ├── footer.php            # Responsive footer with branding & quick links
│   └── auth.php              # Student route protection middleware
